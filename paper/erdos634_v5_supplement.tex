\section{Setting}\label{sec:setting}
Let $u<v<2u$ be coprime positive integers, $d=v-u$, $b=v^2-u^2=d(2u+d)$. The tile is $\Delta=(a,b,c)=(u^2,\,v^2-u^2,\,uv)$ with angles $\alpha$ (opposite $a$), $\beta=\pi-3\alpha$ (opposite $b$), $2\alpha$ (opposite $c$), where $\cos\alpha=v/(2u)$. Let $T(g)$ be the isosceles triangle with legs $g\,u\,b$ and base $g\,v\,b$ (base angles $\alpha$). Its area is $b\,g^2$ times the area of $\Delta$.
Put
\[
g_D=\Big\lfloor \tfrac{u-1}{d}\Big\rfloor+1=\max\{g:(g-1)d<u\}.
\]
\emph{Theorem D$'$} (Lean: \texttt{thmD\_all\_g}, VERIFIED-Lean; the case $g=1$ is Beeson's, 2012): $T(g)$ has no tiling for $g\le g_D$. It is used here as an input and is not reproved.

We use oblique coordinates: $(x,y)$ is the point $x\,e+y\,w$ with $e=1$, $w=e^{i\alpha}$ (an affine image of the plane; incidence, area ratios and disjointness are preserved). The squared length is $|(x,y)|^2=x^2+\frac vu xy+y^2$. Put $s=(v,-u)$. Then $|s|=|(u,-v)|=u$. Take $T(g)$ with vertices $(0,0)$, $(gvb,0)$, $(0,gub)$: the legs are $|(0,gub)|=gub$ and $gb|s|=gub$, the base is $gvb$.

\section{Statements}\label{sec:statements}
\begin{theorem}[Strip, PROVED-hand]\label{thm:strip}
For every coprime $u<v<2u$ and every $g\ge g_D$: any tiling of $T(g)$ extends to a tiling of $T(g+1)$ by adding a region $Z(g)=T(g+1)\setminus T(g)$ tiled by exactly $b(2g+1)$ copies of $\Delta$ (using both chiralities). The tiles of $Z(g)$ are explicit (Section~\ref{sec:proofs}); nothing is assumed about the tiling of $T(g)$.
\end{theorem}
\begin{theorem}[Up-set theorem, PROVED-hand]\label{thm:upset}
Assume coprimality of $u<v<2u$, Theorem D$'$, and that reflections of the tile are allowed. Then $\{g\ge1: T(g)\text{ is tiled}\}=\{g\ge g^*\}$ for some $g^*\in[g_D+1,\infty]$.
\end{theorem}
\begin{proposition}[R1, R2, PROVED-hand]\label{prop:r}
Write $u-1=qd+r$, $0\le r<d$ (so $g_D=q+1$) and $X=(u-1)(v-1)+u^2$. Then
(R1) $X-qb=1+r(2u+d)>0$ and $(q+1)b-X=(d-r)(2u+d)-1>0$, hence $\lceil X/b\rceil=g_D$;
(R2) the strip equation
\begin{equation}\tag{SE}\label{eq:se}
a'u+b'v=g\,b-u^2,\qquad a',b'\in\Z_{\ge0},
\end{equation}
is solvable if $g\ge g_D$ and not solvable if $g\le g_D-1$. Equivalently: a strip with the cap of Lemma~\ref{lem:cap} exists iff $T(g+1)$ is not excluded by D$'$.
\end{proposition}
\begin{theorem}[Collar at $(2,3)$, VERIFIED-Lean]\label{thm:collar23}
The tile is $(4,5,6)$ and $b=5$. For every $g\ge12$, $T(g)$ is tiled by $5g^2$ copies of $(4,5,6)$; hence $5g^2$ is a tile count for every $g\ge12$. The seed $T(12)$, $720$ tiles, is B.~Herdt's layout (kernel-checked), and Theorem~\ref{thm:strip} at $(2,3)$ supplies the induction ($g_D=2$).
\end{theorem}
\begin{remark}
The statement of Theorem~\ref{thm:upset} does not decide whether a tiling exists at a given pair, nor the value of $g^*$. At $(7,10)$ ($b=51$, tile $(49,51,70)$, $g_D=3$) a tiling of $T(130)$ is known (Herdt), so $g^*\le130$ and $T(g)$ is tiled for all $g\ge130$.
\end{remark}

\section{Proofs}\label{sec:proofs}
\paragraph{The strip.}
$T(g+1)\setminus T(g)=\{g\,uvb<ux+vy\le(g+1)uvb,\ x,y\ge0\}$ is a trapezoid; translating by $(0,-gub)$ it becomes the quadrilateral $Z$ with vertices
\[
A=(0,0),\quad B=(0,ub),\quad D=g\,b\,s=(gvb,-gub),\quad C=D+(vb,0).
\]
Here $AB=ub$ (direction $w$), $AD=gub$ (direction $s$), $DC=vb$ (direction $e$), $BC=(g+1)b\,s$, so $AD\parallel BC$.

\begin{lemma}[Cap]\label{lem:cap}
Let $A'$ be a point, $B'=A'+ub\,w$, $D'=A'+u^2s$, $C'=D'+vb\,e$. Then $A'D'C'B'$ is a convex quadrilateral, $B'D'$ is an interior diagonal, and the triangles $A'B'D'$, $B'D'C'$ are $u\Delta$ and $v\Delta$ (SSS). Hence the quadrilateral is tiled by $u^2+v^2$ copies of $\Delta$.
\end{lemma}
\begin{proof}
With $A'=0$: $B'-D'=(-u^2v,\,ub+u^3)=(-u^2v,\,uv^2)=uv\,(-u,v)$, so $|B'D'|=u^2v$. Further $|A'B'|=ub$, $|A'D'|=u^3$, so $A'B'D'$ has sides $(ub,u^3,u^2v)=u(b,a,c)$. Also $|D'C'|=vb$, $|B'D'|=u^2v=va$, and $C'-B'=(v^3,-uv^2)=v^2s$ gives $|B'C'|=uv^2=vc$; so $B'D'C'$ has sides $v(b,a,c)$. The edge vectors $e_1=D'-A'$, $e_2=C'-D'$, $e_3=B'-C'$, $e_4=A'-B'$ have positive oriented cross products $e_1\times e_2=u^3vb$, $e_2\times e_3=uv^3b$, $e_3\times e_4=uv^3b$, $e_4\times e_1=u^3vb$ (multiples of the area form), so the quadrilateral is convex. A $k$-fold scaled copy of $\Delta$ splits into $k^2$ copies (the standard subdivision: $k(k+1)/2$ translates and $k(k-1)/2$ point reflections). \qedhere
\end{proof}
\begin{remark}
That every \emph{minimal} tileable cap has this form is not proved and not used.
\end{remark}

\paragraph{Slices.} A \emph{C-brick} is the parallelogram with sides $p=us$ (length $u^2=a$) and $q=bw$ (length $b$); its diagonal $p-q=(uv,-v^2)=v(u,-v)$ has length $uv=c$, so both halves are congruent to $\Delta$ (SSS; one is the point reflection of the other). A \emph{C-slice} is $u$ bricks stacked along $w$ (a parallelogram with sides $us$, of length $a$ along the strip, and $ub\,w$), $2u$ tiles. A \emph{B-brick} has sides $p=vs$ (length $c$) and $q'=b\,e$ (length $b$); its diagonal is $p-q'=(v^2-b,-uv)=u(u,-v)$ of length $u^2=a$. A \emph{B-slice} is $v$ bricks along $e$, $2v$ tiles. (Both uses of $v^2=u^2+b$ are exactly what makes the diagonals equal $c$ and $a$.)

\paragraph{Assembly (proof of Theorem~\ref{thm:strip}).}
Along the direction $s$ from $A$ to $D$ place $a'$ C-slices, then the cap of Lemma~\ref{lem:cap} (with $A'$ at the end of the C-slices), then $b'$ B-slices. Their lengths along $s$ are $a'u$, $u^2$ and $b'v$ (in units of $s$), and they add up to $g\,b$ if and only if \eqref{eq:se} holds. The interfaces are segments in the directions $w$ (between C-slices and cap) and $e$ (between cap and B-slices); the union of the pieces is $Z$, because the cap's side $B'C'=v^2s$ lies on the line $BC$ and the slices' outer sides lie on $BC$ and on $AD$. The tile count is
\[
2ua'+(u^2+v^2)+2vb'=2(gb-u^2)+u^2+v^2=(2g+1)b,
\]
which equals $\operatorname{area}(Z)/\operatorname{area}(\Delta)=b((g+1)^2-g^2)$. A tiling of $T(g)$ is moved by an isometry (a reflection if needed) onto $T(g)$ at the corner $A$ and shares the segment with $Z$. Solvability of \eqref{eq:se} for $g\ge g_D$ is Proposition~\ref{prop:r}.\hfill$\square$

\paragraph{Proof of Proposition~\ref{prop:r}.}
(R1) $(q+1)b-X=(u-1-r+d)(2u+d)-(2u^2-2u+1+ud-d)=2ud+d^2-2ur-dr-1=(d-r)(2u+d)-1\ge 2u+d-1>0$; the first identity is the same computation.
(R2, solvable) For $g\ge g_D$, $gb-u^2\ge(q+1)b-u^2\ge X-u^2=(u-1)(v-1)$ by (R1), and every integer $n\ge(u-1)(v-1)$ is a nonnegative combination of the coprime integers $u,v$ (Sylvester).
(R2, not solvable) Let $g=g_D-1-m$, $m\ge0$. Then $gb-u^2=uv-k(u+v)$ with $k=1+r+md\ge1$ (using $b=(u+v)d$). If $iu+jv$ were equal to it with $i,j\ge0$, then $(i+k)u+(j+k)v=uv$ with both coefficients $\ge1$; by coprimality $u\mid j+k$ and $v\mid i+k$, so the left side is at least $2uv$, a contradiction. $\square$

\paragraph{Proof of Theorem~\ref{thm:upset}.}
By D$'$ every $g$ with $T(g)$ tiled satisfies $g>g_D$. By Theorem~\ref{thm:strip}, tileability propagates from $g$ to $g+1$. Hence the set is $\{g\ge g^*\}$ with $g^*\ge g_D+1$ (and $g^*=\infty$ if the set is empty). Without D$'$ a tileable $g<g_D$ would not propagate. The construction uses both chiralities, so the theorem is false as stated for direct congruences only. $\square$

\paragraph{Checks (COMPUTED).} The identities were checked by exact integer and rational arithmetic: the cap at 15 pairs (sides, SSS, angles, containment, area, pairwise disjointness); $892$ strips at $12$ pairs (all $g$ from $g_D$ to $40$, first and last solution of \eqref{eq:se}) and $887$ strips (every solution for $g\le20$ at six pairs), $0$ failures; thirty negative controls (a dropped slice, a shifted brick, $a'+1$, a wrong diagonal, a reflected cap, $b'+1$) all fail; R1 and R2 by brute force on all $6115$ coprime pairs with $v\le200$ (least solvable $g$ equals $g_D$ in each case).

\section{Observations (not claimed as results)}\label{sec:obs}
The search engine \texttt{bte} (versions 2.1.2, 2.3.1, 2.4.0; options band:16, la1, cbj1, ng; no symmetry reduction) ended without a tiling on two instances of the base-$\beta$ family: $T_2(4,5)$ with tile $(20,9,25)$, legs $250$ and base $472$ ($46{,}313{,}414$ nodes, reproduced from scratch with the same count), and $T_2(3,5)$ with tile $(15,16,25)$, legs $250$ and base $396$ ($192{,}438{,}239$ nodes; $249{,}950{,}358$ in a second run with a different cache-eviction policy). No certificate was produced; a run with cache eviction is a statement about the engine. They are consistent with the picture that $T_M(e,f)$ is tiled exactly when $M\ge f$; they are not exclusions, are not used above, and are not in the census.
